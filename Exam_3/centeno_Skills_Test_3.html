##Data cleaning

faculty_tidy <- faculty_raw %>%
  pivot_longer(
    cols = c(AvgAssistProfSalary, AvgAssocProfSalary, AvgFullProfSalary),
    names_to = "Rank",
    values_to = "Salary") %>%
  mutate(
    Rank = recode(Rank,
                  AvgAssistProfSalary = "Assistant",
                  AvgAssocProfSalary = "Associate",
                  AvgFullProfSalary = "Full"),
    Rank = factor(Rank, levels = c("Assistant", "Associate", "Full")),
    Tier = factor(Tier),
    State = factor(State))
